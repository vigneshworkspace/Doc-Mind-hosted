# P5: Voice Pipeline (FastRTC + faster-whisper + VibeVoice)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **Prerequisite:** P1 (boot + provider) complete. P3 (RAG pipeline) recommended but not required — voice chat uses `generators.chat_complete()` directly.

**Goal:** Implement real-time voice chat via FastRTC (Gradio's WebRTC library). STT: `faster-whisper` `small` model. TTS: VibeVoice-Realtime-0.5B (Microsoft, MIT) served via an OpenAI-compatible local server on port 8001. Pipeline: browser microphone → WebRTC → FastRTC → faster-whisper STT → LLM (provider fallback chain) → VibeVoice TTS → audio chunks → browser.

**Architecture:** A FastRTC `RealtimeHandler` subclass handles the audio stream. `faster-whisper` runs in-process for low-latency STT. VibeVoice TTS runs as a sidecar server (separate process/container). FastRTC calls it via OpenAI-compatible HTTP. The voice router mounts at `/api/v1/voice` via FastAPI. A WebSocket endpoint `/api/v1/voice/ws` handles signalling for browsers that can't use the Gradio mount.

**Tech Stack:** `fastrtc`, `faster-whisper`, `vibevoice`, Gradio (bundled with FastRTC), NumPy, SoundFile

```python
"""
Speech-to-text using faster-whisper.
Model loaded once on first call and cached in module scope.
"""
import logging
import io
import numpy as np
from typing import Optional

from app.core.config import settings

logger = logging.getLogger(__name__)
_model = None


def _get_model():
    global _model
    if _model is None:
        from faster_whisper import WhisperModel
        device = settings.stt_device
        compute_type = settings.stt_compute_type
        logger.info(f"Loading faster-whisper {settings.stt_model} on {device}/{compute_type}")
        _model = WhisperModel(
            settings.stt_model,
            device=device,
            compute_type=compute_type,
        )
    return _model


def transcribe_audio(audio_bytes: bytes, sample_rate: int = 16000) -> str:
    """
    Transcribe raw PCM audio bytes to text.
    audio_bytes: 16-bit PCM, mono, 16kHz.
    Returns transcription string, or empty string if no speech detected.
    """
    model = _get_model()

    # Convert bytes to float32 numpy array
    audio_np = np.frombuffer(audio_bytes, dtype=np.int16).astype(np.float32) / 32768.0

    segments, info = model.transcribe(
        audio_np,
        language="en",
        beam_size=5,
        vad_filter=True,          # voice activity detection — skip silence
        vad_parameters={"min_silence_duration_ms": 500},
    )

    text = " ".join(seg.text.strip() for seg in segments).strip()
    logger.debug(f"STT: '{text}' (lang={info.language}, prob={info.language_probability:.2f})")
    return text
```

```python
"""
VibeVoice TTS client using the OpenAI-compatible API.
POST /v1/audio/speech — returns WAV bytes.
"""
import httpx
import logging
from app.core.config import settings

logger = logging.getLogger(__name__)


async def text_to_speech(text: str) -> bytes:
    """
    Convert text to speech using VibeVoice server.
    Returns WAV audio bytes.
    Raises httpx.HTTPError on failure.
    """
    url = f"{settings.vibevoice_url}/v1/audio/speech"
    payload = {
        "model": "VibeVoice-Realtime-0.5B",
        "input": text,
        "voice": "alloy",       # VibeVoice 0.5B is single-speaker; voice name ignored
        "response_format": "wav",
        "speed": 1.0,
    }

    async with httpx.AsyncClient(timeout=30.0) as client:
        resp = await client.post(url, json=payload)
        resp.raise_for_status()
        return resp.content


async def is_available() -> bool:
    """Check if VibeVoice server is reachable."""
    try:
        async with httpx.AsyncClient(timeout=3.0) as client:
            resp = await client.get(f"{settings.vibevoice_url}/health")
            return resp.status_code == 200
    except Exception:
        return False
```

## Task 6: Write `handler.py`

**Files:**
- Create: `backend/app/voice/handler.py`

FastRTC's `RealtimeHandler` receives audio chunks from the browser, must return audio chunks back. The pipeline: PCM in → STT → LLM → TTS → PCM out.

- [ ] **Step 1: Write the file**

```python
"""
DocMind FastRTC voice handler.
Receives 16kHz PCM audio, returns 24kHz PCM audio.
"""
import asyncio
import logging
import numpy as np
import io
from typing import Optional

logger = logging.getLogger(__name__)


class DocMindVoiceHandler:
    """
    FastRTC RealtimeHandler subclass.
    Buffers incoming audio, runs STT on silence boundaries,
    calls LLM, then streams TTS back.
    """

    def __init__(self):
        self._audio_buffer = bytearray()
        self._sample_rate = 16000
        self._silence_threshold = 0.01
        self._min_audio_ms = 500       # ignore chunks under 500ms
        self._processing = False

    async def receive(self, frame: tuple) -> None:
        """Called by FastRTC for each incoming audio frame."""
        sample_rate, audio_array = frame  # audio_array: numpy int16
        audio_bytes = audio_array.astype(np.int16).tobytes()
        self._audio_buffer.extend(audio_bytes)

    async def emit(self) -> Optional[tuple]:
        """Called by FastRTC when it needs the next outgoing audio frame."""
        return await self._process_and_respond()

    async def _process_and_respond(self) -> Optional[tuple]:
        if self._processing or len(self._audio_buffer) < self._min_audio_ms * 32:
            return None  # not enough audio or still generating

        self._processing = True
        audio_snapshot = bytes(self._audio_buffer)
        self._audio_buffer.clear()

        try:
            # STT
            from app.voice.stt import transcribe_audio
            text = transcribe_audio(audio_snapshot, self._sample_rate)
            if not text:
                return None

            logger.info(f"Voice STT: '{text}'")

            # LLM
            from app.services.generators import chat_complete
            reply = await chat_complete(
                messages=[{"role": "user", "content": text}],
                context="",
            )
            logger.info(f"Voice LLM reply: '{reply[:80]}'")

            # TTS
            from app.voice.tts_client import text_to_speech, is_available
            if not await is_available():
                logger.warning("VibeVoice unavailable — skipping TTS")
                return None

            wav_bytes = await text_to_speech(reply)

            # Parse WAV → numpy int16
            import soundfile as sf
            audio_np, sr = sf.read(io.BytesIO(wav_bytes), dtype="int16")
            return (sr, audio_np)

        except Exception as e:
            logger.exception(f"Voice pipeline error: {e}")
            return None
        finally:
            self._processing = False
```

```python
"""
Voice router: mounts FastRTC stream + health/status endpoints.
"""
import logging
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends

from app.core.deps import get_current_user
from app.models.orm import User

logger = logging.getLogger(__name__)
router = APIRouter()


@router.get("/health")
async def voice_health():
    """Check if voice pipeline components are available."""
    from app.voice.tts_client import is_available
    tts_ok = await is_available()
    try:
        from faster_whisper import WhisperModel
        stt_available = True
    except ImportError:
        stt_available = False

    return {
        "stt": "available" if stt_available else "not_installed",
        "tts": "available" if tts_ok else "vibevoice_server_unreachable",
    }


@router.websocket("/ws")
async def voice_websocket(
    websocket: WebSocket,
):
    """
    WebSocket voice endpoint.
    Receives binary PCM audio frames (16kHz, mono, int16).
    Sends back binary WAV audio response.

    Message protocol:
    - Client sends binary: raw PCM bytes (int16, 16kHz, mono)
    - Server sends binary: WAV bytes for the response audio
    - Client sends text "END": signals end of utterance (trigger processing)
    - Server sends text "PROCESSING": acknowledgement
    - Server sends text "ERROR:<msg>": on failure
    """
    await websocket.accept()
    audio_buffer = bytearray()

    try:
        while True:
            msg = await websocket.receive()

            if msg["type"] == "websocket.receive":
                if "bytes" in msg and msg["bytes"]:
                    audio_buffer.extend(msg["bytes"])

                elif "text" in msg and msg["text"] == "END":
                    if not audio_buffer:
                        continue

                    await websocket.send_text("PROCESSING")
                    audio_snapshot = bytes(audio_buffer)
                    audio_buffer.clear()

                    try:
                        from app.voice.stt import transcribe_audio
                        text = transcribe_audio(audio_snapshot)
                        if not text:
                            await websocket.send_text("SILENCE")
                            continue

                        from app.services.generators import chat_complete
                        reply = await chat_complete(
                            messages=[{"role": "user", "content": text}],
                            context="",
                        )

                        from app.voice.tts_client import text_to_speech
                        wav_bytes = await text_to_speech(reply)
                        await websocket.send_bytes(wav_bytes)
                        await websocket.send_text(f"TRANSCRIPT:{text}")
                        await websocket.send_text(f"REPLY:{reply}")

                    except Exception as e:
                        logger.exception(f"Voice WS error: {e}")
                        await websocket.send_text(f"ERROR:{str(e)[:200]}")

    except WebSocketDisconnect:
        logger.info("Voice WebSocket disconnected")


def get_fastrtc_app():
    """
    Returns a FastRTC Stream app to mount on the FastAPI app.
    Import is deferred so the server starts even if fastrtc is missing.
    """
    try:
        from fastrtc import Stream, RealtimeHandler
        from app.voice.handler import DocMindVoiceHandler

        class _RTCHandler(RealtimeHandler):
            def __init__(self):
                super().__init__()
                self._inner = DocMindVoiceHandler()

            async def receive(self, frame):
                await self._inner.receive(frame)

            async def emit(self):
                return await self._inner.emit()

        return Stream(_RTCHandler(), mode="send-receive")
    except ImportError:
        logger.warning("fastrtc not installed — FastRTC stream not available")
        return None
```
