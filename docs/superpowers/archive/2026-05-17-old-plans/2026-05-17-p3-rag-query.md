# P3: Retrieval & RAG Query Pipeline

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **Prerequisite:** P1 (boot + provider) and P2 (ingestion) must be complete. Documents must have `parsed_md`, `summary`, `section_summaries`, and `chunks` populated.

**Goal:** Implement the full RAG query pipeline — intent classification, query expansion/decomposition, hybrid retrieval (pgvector + BM25 + RRF), jina-reranker-v3 cross-encoder, grounding threshold check, bounded 2-call agentic fallback, and SSE streaming chat endpoint.

**Architecture:** A new `rag_pipeline.py` orchestrates intent routing → retrieval → rerank → grounding → generation. Three query paths: (1) structural intent → tree traversal, (2) compositional → decompose to sub-queries, (3) single intent → expand to 3-4 rewrites. All paths use `hybrid_retrieve()` (pgvector cosine + Postgres FTS via RRF) → `jina_rerank()` → top-5 chunks + cached doc summary in system prompt. Adaptive gate: docs under 150k tokens and primary provider healthy → stuff full doc (bypass retrieval). Bounded agentic fallback: if top-1 reranker score < 0.40, give LLM 2 tool calls.

**Tech Stack:** SQLAlchemy raw SQL for pgvector + FTS, `sentence-transformers` for jina-reranker-v3, `sse-starlette` for streaming, FastAPI, Pydantic v2

```python
"""
Hybrid retrieval: pgvector cosine + Postgres FTS, merged via Reciprocal Rank Fusion.
"""
from dataclasses import dataclass
from typing import Optional
from sqlalchemy import text
from sqlalchemy.orm import Session

from app.services.embedder import embed_query
from app.core.config import settings


@dataclass
class RetrievedChunk:
    chunk_id: int
    document_id: int
    page_start: Optional[int]
    page_end: Optional[int]
    section_id: Optional[str]
    raw_text: str
    contextualized_text: str
    rrf_score: float


async def hybrid_retrieve(
    query: str,
    db: Session,
    user_id: int,
    document_id: Optional[int] = None,
    top_k: int = None,
) -> list[RetrievedChunk]:
    """
    Run pgvector cosine search + Postgres FTS, then merge via RRF.
    Returns up to rrf_topk chunks sorted by RRF score descending.
    """
    top_k = top_k or settings.retrieve_topk
    rrf_k = settings.rrf_k
    rrf_topk = settings.rrf_topk

    # 1. Vector search
    query_vec = await embed_query(query)
    vec_str = "[" + ",".join(f"{v:.8f}" for v in query_vec) + "]"

    doc_filter = "AND c.document_id = :doc_id" if document_id else ""
    vec_sql = text(f"""
        SELECT c.id, c.document_id, c.page_start, c.page_end,
               c.section_id, c.raw_text, c.contextualized_text,
               ROW_NUMBER() OVER (ORDER BY c.embedding <=> :vec) AS rank
        FROM chunks c
        WHERE c.user_id = :user_id
          {doc_filter}
          AND c.embedding IS NOT NULL
        ORDER BY c.embedding <=> :vec
        LIMIT :topk
    """)
    params = {"vec": vec_str, "user_id": user_id, "topk": top_k}
    if document_id:
        params["doc_id"] = document_id
    vec_rows = db.execute(vec_sql, params).fetchall()

    # 2. BM25 / FTS search
    fts_sql = text(f"""
        SELECT c.id, c.document_id, c.page_start, c.page_end,
               c.section_id, c.raw_text, c.contextualized_text,
               ROW_NUMBER() OVER (ORDER BY ts_rank_cd(c.fts, query) DESC) AS rank
        FROM chunks c,
             websearch_to_tsquery('english', :query) query
        WHERE c.user_id = :user_id
          {doc_filter}
          AND c.fts @@ query
        ORDER BY ts_rank_cd(c.fts, query) DESC
        LIMIT :topk
    """)
    fts_params = {"query": query, "user_id": user_id, "topk": top_k}
    if document_id:
        fts_params["doc_id"] = document_id
    fts_rows = db.execute(fts_sql, fts_params).fetchall()

    # 3. RRF merge
    scores: dict[int, float] = {}
    chunk_data: dict[int, dict] = {}

    for row in vec_rows:
        scores[row.id] = scores.get(row.id, 0.0) + 1.0 / (rrf_k + row.rank)
        chunk_data[row.id] = dict(row._mapping)

    for row in fts_rows:
        scores[row.id] = scores.get(row.id, 0.0) + 1.0 / (rrf_k + row.rank)
        chunk_data.setdefault(row.id, dict(row._mapping))

    # Sort by RRF score, return top rrf_topk
    sorted_ids = sorted(scores, key=lambda x: scores[x], reverse=True)[:rrf_topk]
    return [
        RetrievedChunk(
            chunk_id=cid,
            document_id=chunk_data[cid]["document_id"],
            page_start=chunk_data[cid]["page_start"],
            page_end=chunk_data[cid]["page_end"],
            section_id=chunk_data[cid]["section_id"],
            raw_text=chunk_data[cid]["raw_text"],
            contextualized_text=chunk_data[cid]["contextualized_text"],
            rrf_score=scores[cid],
        )
        for cid in sorted_ids
    ]
```

```python
"""
Cross-encoder reranker: jina-reranker-v3 via sentence-transformers.
Loaded once on first call; cached in module scope.
"""
import logging
from typing import Optional

from app.core.config import settings
from app.services.retrieval import RetrievedChunk

logger = logging.getLogger(__name__)
_model = None


def _get_model():
    global _model
    if _model is None:
        from sentence_transformers import CrossEncoder
        logger.info(f"Loading reranker: {settings.rerank_model}")
        _model = CrossEncoder(settings.rerank_model, max_length=512)
    return _model


def rerank(
    query: str,
    chunks: list[RetrievedChunk],
    top_k: int = None,
) -> list[tuple[RetrievedChunk, float]]:
    """
    Rerank chunks using cross-encoder. Returns (chunk, score) pairs sorted desc.
    Strips 'search_document:' prefix from chunk text before scoring.
    """
    top_k = top_k or settings.rerank_topk
    if not chunks:
        return []

    model = _get_model()

    # Strip Nomic prefix if present
    texts = [
        c.contextualized_text.replace("search_document: ", "", 1)
        for c in chunks
    ]
    pairs = [(query, text) for text in texts]
    scores = model.predict(pairs)

    ranked = sorted(
        zip(chunks, scores.tolist()),
        key=lambda x: x[1],
        reverse=True,
    )
    return ranked[:top_k]


def top1_score(ranked: list[tuple[RetrievedChunk, float]]) -> float:
    """Return the highest reranker score, or 0.0 if empty."""
    return ranked[0][1] if ranked else 0.0
```

```python
"""
Classify query intent: structural | compositional | single.
Uses regex heuristic first; LLM fallback on ambiguity.
"""
import re
from typing import Literal

from app.services.ai import get_provider_with_fallback
from pydantic import BaseModel

Intent = Literal["structural", "compositional", "single"]

_STRUCTURAL = re.compile(
    r"(summarize|summary of|outline|toc|overview of)\s*(chapter|section|ch\.?|sec\.?)\s*\d+|"
    r"(page|p\.?)\s*\d+|"
    r"(chapter|section|ch\.?|sec\.?)\s*\d+\s*(summary|overview|content)",
    re.IGNORECASE,
)

_COMPOSITIONAL = re.compile(
    r"\b(compare|contrast|vs\.?|versus|both|relate|differ|connection between|"
    r"how does .+ relate to|difference between)\b",
    re.IGNORECASE,
)


class _IntentResult(BaseModel):
    intent: Intent


async def classify_intent(query: str) -> Intent:
    """Return structural, compositional, or single."""
    structural_hit = bool(_STRUCTURAL.search(query))
    compositional_hit = bool(_COMPOSITIONAL.search(query))

    if structural_hit and not compositional_hit:
        return "structural"
    if compositional_hit and not structural_hit:
        return "compositional"
    if not structural_hit and not compositional_hit:
        return "single"

    # Ambiguous — ask LLM
    provider = get_provider_with_fallback("gemini", ["groq", "nvidia", "ollama"])
    try:
        result = await provider.structured_output(
            messages=[{
                "role": "user",
                "content": (
                    f"Classify this question as one of: structural, compositional, single.\n"
                    f"- structural: asks directly for a specific section/page/chapter\n"
                    f"- compositional: requires combining info from multiple distinct parts\n"
                    f"- single: a focused question with one primary intent\n\n"
                    f"Question: {query}"
                ),
            }],
            schema=_IntentResult,
            system_prompt="Return JSON only.",
        )
        return result.intent
    except Exception:
        return "single"  # safe fallback
```

```python
"""
Query expansion (single intent) and decomposition (compositional intent).
"""
from pydantic import BaseModel
from typing import List

from app.services.ai import get_provider_with_fallback


class _Rewrites(BaseModel):
    rewrites: List[str]


class _SubQueries(BaseModel):
    sub_queries: List[str]


async def expand_query(query: str) -> list[str]:
    """
    Multi-angle expansion: original + 3 paraphrases.
    Returns list of 4 query strings for parallel retrieval.
    """
    provider = get_provider_with_fallback("gemini", ["groq", "nvidia", "ollama"])
    try:
        result = await provider.structured_output(
            messages=[{
                "role": "user",
                "content": (
                    f"Rewrite this question 3 different ways for retrieval. "
                    f"Preserve the intent but vary wording, focus, and abstraction level.\n\n"
                    f"Question: {query}"
                ),
            }],
            schema=_Rewrites,
            system_prompt="Return JSON only.",
        )
        return [query] + result.rewrites[:3]
    except Exception:
        return [query]  # fallback: original only


async def decompose_query(query: str) -> list[str]:
    """
    Decompose a compositional question into independent sub-questions.
    Returns list of sub-query strings.
    """
    provider = get_provider_with_fallback("gemini", ["groq", "nvidia", "ollama"])
    try:
        result = await provider.structured_output(
            messages=[{
                "role": "user",
                "content": (
                    f"Break this question into independent sub-questions, "
                    f"each retrievable on its own.\n\n"
                    f"Question: {query}"
                ),
            }],
            schema=_SubQueries,
            system_prompt="Return JSON only.",
        )
        return result.sub_queries if result.sub_queries else [query]
    except Exception:
        return [query]
```

```python
"""
RAG pipeline orchestrator.
Entry points: answer() for non-streaming, stream_answer() for SSE.
"""
import logging
from typing import AsyncIterable, Optional

from sqlalchemy.orm import Session
from pydantic import BaseModel
from typing import List

from app.core.config import settings
from app.models.orm import Document
from app.services.ai import get_provider_with_fallback
from app.services.retrieval import hybrid_retrieve, RetrievedChunk
from app.services.reranker import rerank, top1_score
from app.services.intent_classifier import classify_intent
from app.services.query_expander import expand_query, decompose_query

logger = logging.getLogger(__name__)

CITE_INSTRUCTION = (
    "Cite every factual claim with the page number in brackets, e.g. [p.42]. "
    "If no source supports a claim, say 'I could not find this in the provided source.'"
)

REFUSE_TEMPLATE = (
    "I could not find a well-grounded answer for this question in the provided source. "
    "Try rephrasing, or check if the relevant section is in the document."
)


# ── Agentic tool schemas ─────────────────────────────────────────────────────

class _SearchResult(BaseModel):
    chunk_id: int
    page: Optional[int]
    text: str


class _SearchArgs(BaseModel):
    query: str
    top_k: int = 10


class _ReadArgs(BaseModel):
    section_id: str


# ── Helpers ──────────────────────────────────────────────────────────────────

def _render_chunks(chunks: list[RetrievedChunk]) -> str:
    parts = []
    for c in chunks:
        page_str = f"p.{c.page_start}" if c.page_start else "?"
        parts.append(f'<chunk page="{page_str}">\n{c.raw_text}\n</chunk>')
    return "\n\n".join(parts)


def _build_system(doc: Document, chunk_context: str = "") -> str:
    parts = []
    if doc.summary:
        parts.append(f"## Document Summary\n{doc.summary}")
    if chunk_context:
        parts.append(f"## Relevant Sections\n{chunk_context}")
    parts.append(f"## Instructions\n{CITE_INSTRUCTION}")
    return "\n\n".join(parts)


async def _retrieve_and_rerank(
    query: str,
    db: Session,
    user_id: int,
    document_id: Optional[int],
) -> list[tuple[RetrievedChunk, float]]:
    """Single-query retrieve + rerank."""
    chunks = await hybrid_retrieve(query, db, user_id, document_id)
    if not chunks:
        return []
    return rerank(query, chunks)


async def _multi_retrieve(
    queries: list[str],
    db: Session,
    user_id: int,
    document_id: Optional[int],
    query_for_rerank: str,
) -> list[tuple[RetrievedChunk, float]]:
    """Retrieve for multiple queries, dedup by chunk_id, then rerank."""
    seen: set[int] = set()
    all_chunks: list[RetrievedChunk] = []
    for q in queries:
        chunks = await hybrid_retrieve(q, db, user_id, document_id)
        for c in chunks:
            if c.chunk_id not in seen:
                seen.add(c.chunk_id)
                all_chunks.append(c)
    return rerank(query_for_rerank, all_chunks)


# ── Agentic fallback ─────────────────────────────────────────────────────────

async def _agentic_fallback(
    query: str,
    doc: Document,
    db: Session,
    user_id: int,
    document_id: Optional[int],
) -> Optional[str]:
    """
    Give LLM up to agentic_tool_budget tool calls to find better chunks.
    Returns answer string, or None to trigger refusal.
    """
    provider = get_provider_with_fallback("gemini", ["groq", "nvidia", "ollama"])
    budget = settings.agentic_tool_budget
    history = [{"role": "user", "content": query}]
    context_chunks: list[RetrievedChunk] = []

    for _ in range(budget):
        # Ask model for a refined search query
        refine_result = await provider.structured_output(
            messages=history,
            schema=_SearchArgs,
            system_prompt=(
                "The initial retrieval didn't find a confident answer. "
                "Provide a refined search query to find better evidence. "
                "Return JSON with 'query' field."
            ),
        )
        new_chunks = await hybrid_retrieve(
            refine_result.query, db, user_id, document_id, top_k=refine_result.top_k
        )
        context_chunks.extend(new_chunks)
        history.append({
            "role": "assistant",
            "content": f"[searched: {refine_result.query}]",
        })
        history.append({
            "role": "user",
            "content": "Continue. If you have enough context, answer the original question. Otherwise refine again.",
        })

    if not context_chunks:
        return None

    ranked = rerank(query, context_chunks)
    if top1_score(ranked) < 0.20:  # still no good signal
        return None

    system = _build_system(doc, _render_chunks([c for c, _ in ranked]))
    return await provider.chat_completion(
        messages=[{"role": "user", "content": query}],
        system_prompt=system,
    )


# ── Adaptive gate ────────────────────────────────────────────────────────────

def _should_stuff(doc: Document) -> bool:
    """True if document is small enough to stuff full text into context."""
    tc = doc.token_count or 0
    return tc > 0 and tc < settings.stuff_threshold


# ── Main entry points ────────────────────────────────────────────────────────

async def answer(
    query: str,
    messages: list[dict],
    db: Session,
    user_id: int,
    document_id: Optional[int] = None,
) -> str:
    """
    Non-streaming answer. Returns the final response string.
    """
    provider = get_provider_with_fallback("gemini", ["groq", "nvidia", "ollama"])

    doc = None
    if document_id:
        doc = db.query(Document).filter(
            Document.id == document_id, Document.user_id == user_id
        ).first()

    # Adaptive gate: stuff if small enough
    if doc and _should_stuff(doc):
        system = (
            f"{doc.summary or ''}\n\n"
            f"<document>\n{doc.parsed_md or doc.content or ''}\n</document>\n\n"
            f"{CITE_INSTRUCTION}"
        )
        return await provider.chat_completion(messages=messages, system_prompt=system)

    # Intent classify
    intent = await classify_intent(query)
    logger.debug(f"Intent: {intent}")

    if intent == "structural":
        # Section lookup via outline + section summaries
        section_text = _resolve_structural(query, doc)
        system = _build_system(doc or _empty_doc(), section_text)
        return await provider.chat_completion(messages=messages, system_prompt=system)

    if intent == "compositional":
        sub_queries = await decompose_query(query)
        ranked = await _multi_retrieve(sub_queries, db, user_id, document_id, query)
    else:
        # single — expand
        expanded = await expand_query(query)
        ranked = await _multi_retrieve(expanded, db, user_id, document_id, query)

    # Grounding check
    score = top1_score(ranked)
    logger.debug(f"Top-1 reranker score: {score:.3f}")

    if score < settings.grounding_threshold:
        # Bounded agentic fallback
        fallback_answer = await _agentic_fallback(query, doc or _empty_doc(), db, user_id, document_id)
        if fallback_answer is None:
            return REFUSE_TEMPLATE
        return fallback_answer

    top_chunks = [c for c, _ in ranked[: settings.rerank_topk]]
    system = _build_system(doc or _empty_doc(), _render_chunks(top_chunks))
    return await provider.chat_completion(messages=messages, system_prompt=system)


async def stream_answer(
    query: str,
    messages: list[dict],
    db: Session,
    user_id: int,
    document_id: Optional[int] = None,
) -> AsyncIterable[str]:
    """
    SSE streaming answer. Yields token chunks.
    Falls back to non-streaming for agentic fallback path.
    """
    provider = get_provider_with_fallback("gemini", ["groq", "nvidia", "ollama"])

    doc = None
    if document_id:
        doc = db.query(Document).filter(
            Document.id == document_id, Document.user_id == user_id
        ).first()

    # Adaptive gate
    if doc and _should_stuff(doc):
        system = (
            f"{doc.summary or ''}\n\n"
            f"<document>\n{doc.parsed_md or doc.content or ''}\n</document>\n\n"
            f"{CITE_INSTRUCTION}"
        )
        async for chunk in provider.stream_completion(messages=messages, system_prompt=system):
            yield chunk
        return

    intent = await classify_intent(query)

    if intent == "structural":
        section_text = _resolve_structural(query, doc)
        system = _build_system(doc or _empty_doc(), section_text)
    elif intent == "compositional":
        sub_queries = await decompose_query(query)
        ranked = await _multi_retrieve(sub_queries, db, user_id, document_id, query)
        if top1_score(ranked) < settings.grounding_threshold:
            fallback = await _agentic_fallback(query, doc or _empty_doc(), db, user_id, document_id)
            yield fallback or REFUSE_TEMPLATE
            return
        system = _build_system(doc or _empty_doc(), _render_chunks([c for c, _ in ranked[:settings.rerank_topk]]))
    else:
        expanded = await expand_query(query)
        ranked = await _multi_retrieve(expanded, db, user_id, document_id, query)
        if top1_score(ranked) < settings.grounding_threshold:
            fallback = await _agentic_fallback(query, doc or _empty_doc(), db, user_id, document_id)
            yield fallback or REFUSE_TEMPLATE
            return
        system = _build_system(doc or _empty_doc(), _render_chunks([c for c, _ in ranked[:settings.rerank_topk]]))

    async for chunk in provider.stream_completion(messages=messages, system_prompt=system):
        yield chunk


def _resolve_structural(query: str, doc: Optional[Document]) -> str:
    """Extract section content from doc.section_summaries for structural queries."""
    if not doc or not doc.section_summaries:
        return ""
    import re
    # Look for chapter/section number in query
    m = re.search(r"(chapter|section|ch|sec)\.?\s*(\d+)", query, re.IGNORECASE)
    if not m:
        return ""
    target = int(m.group(2))
    # section_summaries is list of {section_title, summary}; use ordinal position
    summaries = doc.section_summaries or []
    if 0 < target <= len(summaries):
        s = summaries[target - 1]
        return f"**{s.get('section_title', '')}**\n{s.get('summary', '')}"
    return ""


def _empty_doc() -> Document:
    d = Document()
    d.summary = ""
    d.section_summaries = []
    d.parsed_md = ""
    d.content = ""
    d.token_count = 0
    return d
```

```python
from fastapi import APIRouter, Depends
from fastapi.responses import StreamingResponse
from sqlalchemy.orm import Session
from typing import List
from app.core.deps import get_db, get_current_user
from app.models.orm import User, ChatHistory
from app.schemas.schemas import ChatCompleteRequest, ChatCompleteResponse, ChatHistoryOut
from app.services.rag_pipeline import answer, stream_answer

router = APIRouter()


@router.post("/complete", response_model=ChatCompleteResponse)
async def complete(
    req: ChatCompleteRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    messages = [{"role": m.role, "content": m.content} for m in req.messages]
    query = messages[-1]["content"] if messages else ""

    reply = await answer(
        query=query,
        messages=messages,
        db=db,
        user_id=current_user.id,
        document_id=req.document_id,
    )

    # Persist
    chat_record = ChatHistory(
        user_id=current_user.id,
        document_id=req.document_id,
        messages=messages + [{"role": "assistant", "content": reply}],
    )
    db.add(chat_record)
    db.commit()

    return ChatCompleteResponse(content=reply, model="docmind-rag")


@router.post("/stream")
async def stream_complete(
    req: ChatCompleteRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    messages = [{"role": m.role, "content": m.content} for m in req.messages]
    query = messages[-1]["content"] if messages else ""

    async def event_stream():
        async for chunk in stream_answer(
            query=query,
            messages=messages,
            db=db,
            user_id=current_user.id,
            document_id=req.document_id,
        ):
            yield f"data: {chunk}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(event_stream(), media_type="text/event-stream")


@router.get("/history", response_model=List[ChatHistoryOut])
async def get_chat_history(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    return (
        db.query(ChatHistory)
        .filter(ChatHistory.user_id == current_user.id)
        .order_by(ChatHistory.created_at.desc())
        .all()
    )
```
